cmake_minimum_required(VERSION 3.22)
project(dronedream_camera_clock LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(dronedream-camera-clock SHARED camera_clock.cpp)
target_link_libraries(dronedream-camera-clock PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
target_compile_options(dronedream-camera-clock PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(capture-clock-test capture_clock_test.cpp)
add_test(NAME capture-clock-contract COMMAND capture-clock-test)
